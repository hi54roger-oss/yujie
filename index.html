在 C++ 中處理 Discord 機器人需要極高的細緻度。我採用了模組化設計，將核心邏輯與指令實作分離，並利用 DPP 的 Lambda 表達式簡化事件註冊。

  • 高效的 Lambda 事件回調處理
  • 自定義的指令路由選擇器
  • 異步 HTTP 請求與 JSON 解析
main.cpp 
#include <dpp/dpp.h>

int main() {
    dpp::cluster bot("BOT_TOKEN");

    bot.on_log(dpp::utility::log_stderr);

    bot.on_slashcommand([](const auto& event) {
        if (event.command.get_command_name() == "ai_ask") {
            // 爬蟲與 AI 處理邏輯
            event.reply("思考中...");
            fetch_ai_response(event);
        }
    });

    bot.start(dpp::st_wait);
    return 0;
}
